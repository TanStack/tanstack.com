let instance
let requests = 0
export default {
  async fetch(request, env, ctx) {
    instance ??= crypto.randomUUID()
    const count = ++requests
    const started = performance.now()
    const path = new URL(request.url).pathname
    let response
    let phaseMs = 0
    if (path === '/cache') {
      const t = performance.now()
      await caches.default.match('https://blob-storage-cache.tanstack.internal/diagnostic-missing-key')
      phaseMs = performance.now() - t
      response = new Response('cache read complete')
    } else if (path === '/r2') {
      const t = performance.now()
      await env.GITHUB_CONTENT_CACHE.head('diagnostic-missing-key')
      phaseMs = performance.now() - t
      response = new Response('r2 read complete')
    } else if (path === '/docs') {
      response = new Response('not supported', {status:404})
    } else {
      response = new Response('ok')
    }
    const headers = new Headers(response.headers)
    headers.set('Cache-Control', 'private, no-store')
    headers.set('X-Probe-Instance', instance)
    headers.set('X-Probe-Request', String(count))
    headers.set('X-Probe-Phase-Ms', String(phaseMs))
    console.log(JSON.stringify({logType:'docs_cold_probe',instance,count,path,ray:request.headers.get('cf-ray'),phaseMs,headersMs:performance.now()-started}))
    return new Response(response.body, {status:response.status,headers})
  }
}
